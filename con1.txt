	local v2 = tostring(p1 or "")
	local v3 = v1.BaseRarityStars or {}

	local v5 = v1.GodStarsByRarity or {}

	local v9 = tostring(v1.OmniscientRarity or "Omniscient")

		return v1.OmniscientStars or 5, v1.EvolvedColor or Color3.fromRGB(255, 255, 255), v1.OmniscientGradient or "Omniscient"
	end

		return v5[v2] or (v1.GodStars or 5), v1.EvolvedColor or Color3.fromRGB(255, 255, 255), v1.GodGradient or "God"
